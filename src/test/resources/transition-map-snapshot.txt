BANK_TRX
  [initial] -> [NEW]
  ENRICHED -> [MANUAL_REVIEW, PAIRED, POSTING_READY]
  ERROR -> [NEW]
  MANUAL_REVIEW -> [ENRICHED, NEW, POSTING_READY]
  NEW -> [PROCESSING]
  PAIRED -> [POSTED]
  POSTING_READY -> [POSTED]
  PROCESSING -> [ENRICHED, ERROR, MANUAL_REVIEW]

ENRICHMENT
  [initial] -> [NEW]
  ADJUSTED -> [ENRICHED, IN_REVIEW, READY, SUPERSEDED]
  CONFIRMED -> []
  ENRICHED -> [ADJUSTED, IN_REVIEW, MANUAL_REVIEW, PAIRED, READY, SUPERSEDED]
  ERROR -> [MANUAL_REVIEW, NEW]
  IN_REVIEW -> [ADJUSTED, ENRICHED, READY, SUPERSEDED]
  MANUAL_REVIEW -> [ENRICHED, NEW, READY]
  NEW -> [PROCESSING]
  PAIRED -> [MANUAL_REVIEW, READY]
  PROCESSING -> [ENRICHED, ERROR, MANUAL_REVIEW]
  READY -> [CONFIRMED, ENRICHED, IN_REVIEW, SUPERSEDED]
  SUPERSEDED -> []

EXCEPTION
  [initial] -> [OPEN]
  DISMISSED -> []
  IN_PROGRESS -> [DISMISSED, RESOLVED]
  OPEN -> [DISMISSED, IN_PROGRESS]
  RESOLVED -> []

PAIR
  [initial] -> [AUTO_ACCEPTED, PENDING_REVIEW]
  AUTO_ACCEPTED -> []
  CONFIRMED -> []
  PENDING_REVIEW -> [CONFIRMED, REJECTED]
  REJECTED -> []

POSTING_OPERATION
  [initial] -> [PENDING]
  ERROR -> [PENDING, REVOKED]
  PENDING -> [POSTING, REVOKED]
  POSTED -> []
  POSTING -> [ERROR, POSTED]
  REVOKED -> []

SECU_TRX
  [initial] -> [NEW]
  ENRICHED -> [MANUAL_REVIEW, PAIRED, UNMATCHED]
  ERROR -> [NEW]
  MANUAL_REVIEW -> [ENRICHED, NEW, PAIRED]
  NEW -> [PROCESSING]
  PAIRED -> [POSTED]
  PROCESSING -> [ENRICHED, ERROR, MANUAL_REVIEW]
  UNMATCHED -> [MANUAL_REVIEW, PAIRED]

STATEMENT
  [initial] -> [NEW]
  CONSOLIDATED -> [ENRICHED, ERROR]
  CONSOLIDATING -> [CONSOLIDATED, ERROR]
  ENRICHED -> [POSTED]
  ERROR -> [NEW]
  IMPORTED -> [CONSOLIDATING]
  IMPORTING -> [ERROR, IMPORTED]
  NEW -> [IMPORTING]
